Get the data

Read the frozen release directly — from R, Python, a browser, ERDDAP or any DuckDB. No server, no credentials.

library(calcofi4r)
con <- cc_get_db()     # latest release
tbl(con, "obs_env") |>
  filter(measurement_type == "temperature",
         depth_m <= 10)
import calcofi4py as cc
con = cc.cc_get_db()    # latest release
con.sql("""SELECT * FROM obs_env
  WHERE measurement_type = 'temperature'
    AND depth_m <= 10""").df()
-- db-query: DuckDB in your browser, no server
SELECT cruise_key, avg(value) AS temperature
FROM obs_env
WHERE measurement_type = 'temperature'
GROUP BY cruise_key
